# mini-9pi's SD card (kernels/9pi's make card): what the Raspberry Pi's
# firmware reads from the FAT partition before it starts a kernel
# (kernels/firmware/README.md).

# the firmware's own memory, the least: the cut-down firmware (start_cd.elf)
gpu_mem=16

# the Pi 1: mini-9pi with ix's programs in its image (make ix), loaded
# at 0x8000. The last kernel= line, and under no [pi1]: this firmware
# (2015) may not know the sections, and would take them all.
kernel=mini9pi1.img

# The processor overclocked (it is 700 MHz), by the Foundation's
# settings for a Pi 1: one block of the four below, the others
# commented out. force_turbo: at that speed always, not only when the
# system asks (mini-9pi does not ask). The kernel says the speed it
# measures, before the prompt ("cpu: 800 MHz") and in /dev/cputype. If
# the board does not start, or programs die: a slower block, or none.

# Modest: no more voltage. The author's Pi1, 2026-10-08: "looks good,
# slightly faster indeed", and the kernel's 800.
arm_freq=800
core_freq=250
sdram_freq=400
over_voltage=0
force_turbo=1
