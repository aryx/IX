# mini-9pi's SD card (kernels/9pi's make card): what the Raspberry Pi's
# firmware reads from the FAT partition before it starts a kernel
# (kernels/firmware/README.md).

# the firmware's own memory, the least: the cut-down firmware (start_cd.elf)
gpu_mem=16

# the Pi 1: mini-9pi with ix's programs in its image (make ix), loaded
# at 0x8000. The last kernel= line, and under no [pi1]: this firmware
# (2015) may not know the sections, and would take them all.
kernel=mini9pi1.img

# the processor at 800 MHz (it is 700): the Foundation's "Modest"
# setting for a Pi 1, with no more voltage (the board's warranty bit is
# set only by more voltage with force_turbo). force_turbo: at that
# speed always, not only when the system asks (mini-9pi does not ask).
# Faster ones, each to try on the board (its power, its card): Medium
# arm_freq=900 sdram_freq=450 over_voltage=2; High 950, 450, 6; Turbo
# 1000, core_freq=500, sdram_freq=600, over_voltage=6 (known to
# corrupt SD cards). The author's Pi1 with these lines, 2026-10-08:
# "looks good, slightly faster indeed". The kernel's "cpu0:" line says
# the speed it measures. If a board does not start, take them out.
arm_freq=800
core_freq=250
sdram_freq=400
over_voltage=0
force_turbo=1
